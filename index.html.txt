/* ===================== 工具 ===================== */
function norm(v) {
  if (v === null || v === undefined) return '';
  return String(v).replace(/\u00a0/g, ' ').replace(/\s+/g, ' ').trim();
}
function isNumeric(s) { return s !== '' && !isNaN(Number(s)); }
function cellEqual(a, b) {
  const sa = norm(a), sb = norm(b);
  if (sa === sb) return true;
  if (isNumeric(sa) && isNumeric(sb) && Number(sa) === Number(sb)) return true;
  return false;
}
function keyOf(row, idxs) { return idxs.map(i => norm(row[i])).join('\u0001'); }
function keyText(row, idxs) { return idxs.map(i => norm(row[i])).join(' | '); }
function rowText(row, idxs) {
  if (!idxs) return row.map(v => norm(v)).join(' | ');
  return idxs.map(i => norm(row[i])).join(' | ');
}
function escHtml(s) {
  return norm(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}
function escAttr(s) {
  return String(s == null ? '' : s).replace(/&/g, '&amp;').replace(/"/g, '&quot;').replace(/</g, '&lt;');
}
function escXml(s) {
  return norm(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&apos;');
}
/* 关键：单侧多条件筛选 */
function matchRow(row, side) {
  const kw = side === 1 ? S.kw1 : S.kw2;
  const set = side === 1 ? S.set1 : S.set2;
  const g = side === 1 ? S.global1 : S.global2;
  for (const ci in kw) {
    const v = String(row[+ci] == null ? '' : row[+ci]).toLowerCase();
    if (v.indexOf(kw[ci]) < 0) return false;
  }
  for (const ci in set) {
    if (!set[ci].has(norm(row[+ci]))) return false;
  }
  if (g) {
    const hit = row.some(v => String(v == null ? '' : v).toLowerCase().indexOf(g) >= 0);
    if (!hit) return false;
  }
  return true;
}

async function loadFile(file, side) {
  try {
    const buf = await file.arrayBuffer();
    const wb = XLSX.read(buf, { type: 'array' });
    if (side === 1) { S.wb1 = wb; S.name1 = file.name; S.sheet1 = wb.SheetNames[0]; }
    else            { S.wb2 = wb; S.name2 = file.name; S.sheet2 = wb.SheetNames[0]; }
    updateUploadUI(side);
    if (S.wb1 && S.wb2) { $('settingsCard').hidden = false; fillSheetSelects(); parseBoth(false, false); }
  } catch (err) {
    alert('文件解析失败：' + err.message);
  }
}

function parseSheet(wb, sheetName, headerRow) {
  const ws = wb.Sheets[sheetName];
  const aoa = XLSX.utils.sheet_to_json(ws, { header: 1, defval: '', raw: false, blankrows: false });
  const hIdx = Math.max(0, headerRow - 1);
  const rawHeader = aoa[hIdx] || [];
  const headers = [], count = {};
  rawHeader.forEach((h, i) => {
    let name = norm(h) || ('列' + (i + 1));
    if (count[name]) { count[name]++; name = name + '_' + count[name]; }
    else count[name] = 1;
    headers.push(name);
  });
  const rows = aoa.slice(hIdx + 1)
    .filter(r => r.some(c => norm(c) !== ''))
    .map(r => {
      const arr = [];
      for (let i = 0; i < headers.length; i++) arr.push(r[i] === undefined || r[i] === null ? '' : r[i]);
      return arr;
    });
  return { headers, rows };
}

    if (g1) g1.value = ''; if (g2) g2.value = '';
  }

  if (keepSelection) {

  } else {
    autoPair();

  }

  renderPairList();
  renderPreview(1);
  renderPreview(2);
}

function autoPair() {
  S.pairs = [];
  const h1 = S.t1.headers, h2 = S.t2.headers;
  const used2 = new Set();
  h1.forEach((name, i1) => {
    const i2 = h2.indexOf(name);
    if (i2 >= 0 && !used2.has(i2)) { S.pairs.push({ i1, i2 }); used2.add(i2); }
  });
}
function addColumnIfMissing(side, ci) {
  if (S.pairs.some(p => (side === 1 ? p.i1 : p.i2) === ci)) return;
  const myHeaders    = side === 1 ? S.t1.headers : S.t2.headers;
  const otherHeaders = side === 1 ? S.t2.headers : S.t1.headers;
  const usedOther = new Set(S.pairs.map(p => side === 1 ? p.i2 : p.i1));
  const name = myHeaders[ci];
  let otherIdx = otherHeaders.indexOf(name);
  if (otherIdx >= 0 && usedOther.has(otherIdx)) otherIdx = -1;
  if (otherIdx < 0) {
    for (let i = 0; i < otherHeaders.length; i++) if (!usedOther.has(i)) { otherIdx = i; break; }
  }
  if (otherIdx < 0) otherIdx = 0;
  if (side === 1) S.pairs.push({ i1: ci, i2: otherIdx });
  else            S.pairs.push({ i1: otherIdx, i2: ci });
}
function toggleColumn(side, ci) {
  const idx = S.pairs.findIndex(p => (side === 1 ? p.i1 : p.i2) === ci);
  if (idx >= 0) S.pairs.splice(idx, 1); else addColumnIfMissing(side, ci);
  renderPairList(); renderPreview(1); renderPreview(2);
}

      <select class="pair-sel2">${S.t2.headers.map((h, i) =>
        `<option value="${i}" ${i === p.i2 ? 'selected' : ''}>${escHtml(h)}</option>`).join('')}</select>
      <button class="pair-del">删除</button>

}

function doCompare() {
  const keyNames = getChecked('key');
  if (!keyNames.length) { alert('请至少选择一个主键列'); return; }
  if (!S.pairs.length) { alert('请至少添加一对对比列'); return; }

  const h1 = S.t1.headers, h2 = S.t2.headers;
  const keyIdx1 = keyNames.map(n => h1.indexOf(n));
  const keyIdx2 = keyNames.map(n => h2.indexOf(n));
  if (keyIdx1.some(i => i < 0) || keyIdx2.some(i => i < 0)) { alert('主键列在两表中无法完全对应'); return; }

  const matchSet1 = new Set();
  S.t1.rows.forEach((row, ri) => { if (matchRow(row, 1)) matchSet1.add(ri); });
  const matchSet2 = new Set();
  S.t2.rows.forEach((row, ri) => { if (matchRow(row, 2)) matchSet2.add(ri); });

  const actualSel1 = new Set([...S.sel1].filter(i => matchSet1.has(i)));
  const actualSel2 = new Set([...S.sel2].filter(i => matchSet2.has(i)));

  if (!actualSel1.size) { alert('表格1 没有参与对比的行（请检查筛选条件与勾选）'); return; }
  if (!actualSel2.size) { alert('表格2 没有参与对比的行（请检查筛选条件与勾选）'); return; }

  S.result = runCompare(S.t1, S.t2, keyIdx1, keyIdx2, S.pairs, actualSel1, actualSel2);
  renderResult();
}

function runCompare(t1, t2, keyIdx1, keyIdx2, pairs, sel1, sel2) {
  const marks1 = t1.rows.map(() => new Array(t1.headers.length).fill(''));
  const marks2 = t2.rows.map(() => new Array(t2.headers.length).fill(''));
  const records = [];
  const summary = { total1: sel1.size, total2: sel2.size, same: 0, add: 0, del: 0, chgRows: 0, chgCells: 0 };
  const rowPairs = [];

  const idx2 = new Map();
  t2.rows.forEach((row, i) => {
    if (!sel2.has(i)) return;
    const k = keyOf(row, keyIdx2);
    if (!idx2.has(k)) idx2.set(k, []);
    idx2.get(k).push(i);
  });
  const used2 = new Set();

  t1.rows.forEach((r1, i1) => {
    if (!sel1.has(i1)) return;
    const k = keyOf(r1, keyIdx1);
    const queue = idx2.get(k);
    const i2 = (queue && queue.length) ? queue.shift() : undefined;

    if (i2 === undefined) {
      pairs.forEach(p => { if (norm(r1[p.i1]) !== '') marks1[i1][p.i1] = 'add'; });
      summary.add++;
      rowPairs.push({ i1, i2: undefined });
      records.push({
        type: '增加', key: keyText(r1, keyIdx1), row1: i1 + 1, row2: '',
        field: '整行', v1: rowText(r1, pairs.map(p => p.i1)), v2: '', note: '表格1有，表格2没有'
      });
      return;
    }

    used2.add(i2);
    rowPairs.push({ i1, i2 });
    const r2 = t2.rows[i2];
    let diffCount = 0;

    pairs.forEach(p => {
      const a = r1[p.i1], b = r2[p.i2];
      const aEmpty = norm(a) === '';
      const bEmpty = norm(b) === '';
      if (cellEqual(a, b)) {
        if (!aEmpty) marks1[i1][p.i1] = 'same';
        if (!bEmpty) marks2[i2][p.i2] = 'same';
      } else {
        diffCount++;
        if (!aEmpty) marks1[i1][p.i1] = 'chg';
        if (!bEmpty) marks2[i2][p.i2] = 'chg';
        records.push({
          type: '变化', key: keyText(r1, keyIdx1), row1: i1 + 1, row2: i2 + 1,
          field: t1.headers[p.i1] + (t1.headers[p.i1] !== t2.headers[p.i2] ? ' → ' + t2.headers[p.i2] : ''),
          v1: a, v2: b, note: '内容不同'
        });
      }
    });

    if (diffCount === 0) {
      summary.same++;
      records.push({
        type: '相同', key: keyText(r1, keyIdx1), row1: i1 + 1, row2: i2 + 1,
        field: '', v1: '', v2: '', note: '整行一致'
      });
    } else {
      summary.chgRows++;
      summary.chgCells += diffCount;
    }
  });

  t2.rows.forEach((r2, i2) => {
    if (!sel2.has(i2) || used2.has(i2)) return;
    pairs.forEach(p => { if (norm(r2[p.i2]) !== '') marks2[i2][p.i2] = 'del'; });
    summary.del++;
    rowPairs.push({ i1: undefined, i2 });
    records.push({
      type: '减少', key: keyText(r2, keyIdx2), row1: '', row2: i2 + 1,
      field: '整行', v1: '', v2: rowText(r2, pairs.map(p => p.i2)), note: '表格2有，表格1没有'
    });
  });

  return {
    marks1, marks2, records, summary, pairs: rowPairs,
    pairsDef: pairs.map(p => ({ i1: p.i1, i2: p.i2 })),
    sel1, sel2
  };
}

/* ===================== 结果筛选逻辑 ===================== */
function rowStatus(marksRow, len) {
  const s = { same: false, diff: false, add: false, del: false, chg: false };
  for (let ci = 0; ci < len; ci++) {
    const m = marksRow[ci];
    if (m === 'same') s.same = true;
    else if (m === 'add') { s.add = true; s.diff = true; }
    else if (m === 'del') { s.del = true; s.diff = true; }
    else if (m === 'chg') { s.chg = true; s.diff = true; }
  }
  if (!s.diff) s.same = true; // 无任何差异标记 → 视为相同行
  return s;
}
function passRowFilter(s, text) {
  const ty = S.ui.resultType;
  if (ty === 'same' && s.diff) return false;
  if (ty === 'diff' && !s.diff) return false;
  if (ty === 'chg' && !s.chg) return false;
  if (ty === 'add' && !s.add) return false;
  if (ty === 'del' && !s.del) return false;
  const kw = S.ui.resultSearch.toLowerCase();
  if (kw && text.toLowerCase().indexOf(kw) < 0) return false;
  return true;
}

function refreshResultPanes() {
  const { t1, t2, result } = S;
  const { marks1, marks2, records } = result;
  let shown = 0;

  const t1html = renderCmpTable(t1, marks1, 1, result.sel1);
  const t2html = renderCmpTable(t2, marks2, 2, result.sel2);
  const sbsHtml = renderSbs();

  $('pane-t2').innerHTML = `<div class="table-wrap">${t2html.html}</div>`;
  $('pane-sbs').innerHTML = `<div class="table-wrap">${sbsHtml.html}</div>`;
  $('pane-rec').innerHTML = `<div class="table-wrap">${recHtml.html}</div>`;

  shown = recHtml.count;
  $('resultInfo').textContent = `当前筛选：${shown} 条记录 / ${records.length} 条总记录`;
}

function renderCmpTable(t, marks, side, sel) {
  const clsMap = side === 1
    ? { same: 'c-same', add: 'c-red', chg: 'c-red' }
    : { same: 'c-same', del: 'c-yellow', chg: 'c-yellow' };

  let html = '<table class="cmp"><thead><tr>';
  html += '<th class="rowno-h">行号</th>';
  t.headers.forEach(h => { html += `<th>${escHtml(h)}</th>`; });
  html += '</tr></thead><tbody>';

  t.rows.forEach((row, ri) => {
    if (!sel.has(ri)) return;
    const status = rowStatus(marks[ri], t.headers.length);
    const text = row.map(v => norm(v)).join(' | ');
    if (!passRowFilter(status, text)) return;
    html += '<tr>';
    html += `<td class="rowno">${ri + 1}</td>`;
    row.forEach((v, ci) => {
      const m = marks[ri][ci];
      const cls = (m && norm(v) !== '') ? clsMap[m] : '';
      html += `<td class="${cls}">${escHtml(v)}</td>`;
    });
    html += '</tr>';
  });
  html += '</tbody></table>';
  return { html };
}

function renderSbs() {
  const { t1, t2, result } = S;
  const { marks1, marks2, pairsDef } = result;
  if (!pairsDef) return { html: '' };

  let html = '<table class="sbs"><thead>';
  html += '<tr class="group-row">';
  html += `<th class="sbs-group" colspan="${t1.headers.length}">表格1</th>`;
  html += `<th class="sbs-group t2" colspan="${t2.headers.length}">表格2</th>`;
  html += '</tr><tr>';
  t1.headers.forEach(h => { html += `<th>${escHtml(h)}</th>`; });
  t2.headers.forEach(h => { html += `<th>${escHtml(h)}</th>`; });
  html += '</tr></thead><tbody>';

  const kw = S.ui.resultSearch.toLowerCase();
  result.pairs.forEach(pr => {
    const row1 = pr.i1 !== undefined ? t1.rows[pr.i1] : null;
    const row2 = pr.i2 !== undefined ? t2.rows[pr.i2] : null;

    let diff = false, chg = false, add = false, del = false;
    if (pr.i2 === undefined) { diff = true; add = true; }
    else if (pr.i1 === undefined) { diff = true; del = true; }
    else {
      pairsDef.forEach(p => {
        if (marks1[pr.i1][p.i1] === 'chg') { diff = true; chg = true; }
      });
    }

    if (S.ui.sbsDiffOnly && !diff) return;

    const ty = S.ui.resultType;
    if (ty === 'same' && diff) return;
    if (ty === 'diff' && !diff) return;
    if (ty === 'chg' && !chg) return;
    if (ty === 'add' && !add) return;
    if (ty === 'del' && !del) return;

    if (kw) {
      const text = [
        ...(row1 ? row1.map(v => norm(v)) : []),
        ...(row2 ? row2.map(v => norm(v)) : [])
      ].join(' ').toLowerCase();
      if (text.indexOf(kw) < 0) return;
    }

    html += '<tr>';
    t1.headers.forEach((_, ci) => {
      const v = row1 ? row1[ci] : '';
      let cls = '';
      if (row1 && norm(v) !== '') {
        const m = marks1[pr.i1][ci];
        if (m === 'same') cls = 'c-same';
        else if (m === 'add' || m === 'chg') cls = 'c-red';
      }
      html += `<td class="${cls}">${escHtml(v)}</td>`;
    });
    t2.headers.forEach((_, ci) => {
      const v = row2 ? row2[ci] : '';
      let cls = '';
      if (row2 && norm(v) !== '') {
        const m = marks2[pr.i2][ci];
        if (m === 'same') cls = 'c-same';
        else if (m === 'del' || m === 'chg') cls = 'c-yellow';
      }
      html += `<td class="${cls}">${escHtml(v)}</td>`;
    });
    html += '</tr>';
  });
  html += '</tbody></table>';
  return { html };
}

function renderRecords(records) {
  const badgeMap = { '增加': 'badge-add', '减少': 'badge-del', '相同': 'badge-same', '变化': 'badge-chg' };
  const ty = S.ui.resultType;
  const kw = S.ui.resultSearch.toLowerCase();
  const typeMap = { same: '相同', chg: '变化', add: '增加', del: '减少' };

  let html = '<table class="rec"><thead><tr>';
  ['序号', '类型', '主键', '表格1行号', '表格2行号', '字段', '表格1值', '表格2值', '说明']
    .forEach(h => { html += `<th>${h}</th>`; });
  html += '</tr></thead><tbody>';

  let count = 0;
  records.forEach((r, i) => {
    if (ty !== 'all') {
      const target = typeMap[ty];
      if (target && r.type !== target) return;
      if (ty === 'diff' && r.type === '相同') return;
    }
    if (kw) {
      const text = [r.key, r.field, r.v1, r.v2, r.note].join(' ').toLowerCase();
      if (text.indexOf(kw) < 0) return;
    }
    count++;
    html += `<tr>
      <td>${i + 1}</td>
      <td><span class="badge ${badgeMap[r.type] || ''}">${r.type}</span></td>
      <td>${escHtml(r.key)}</td>
      <td>${r.row1}</td>
      <td>${r.row2}</td>
      <td>${escHtml(r.field)}</td>
      <td>${escHtml(r.v1)}</td>
      <td>${escHtml(r.v2)}</td>
      <td>${escHtml(r.note)}</td>
    </tr>`;
  });
  if (!count) html += '<tr><td colspan="9" style="text-align:center;color:#999;padding:20px">没有符合筛选条件的记录</td></tr>';
  html += '</tbody></table>';
  return { html, count };
}

function exportReport() {
  if (!S.result) { alert('请先执行比对'); return; }
  const xml = buildWorkbookXml();
  const blob = new Blob(['\ufeff' + xml], { type: 'application/vnd.ms-excel;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = '表格比对报告_' + dateStamp() + '.xls';
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  setTimeout(() => URL.revokeObjectURL(url), 2000);
}
function dateStamp() {
  const d = new Date(); const p = n => String(n).padStart(2, '0');
  return d.getFullYear() + p(d.getMonth() + 1) + p(d.getDate()) + '_' + p(d.getHours()) + p(d.getMinutes()) + p(d.getSeconds());
}

function buildWorkbookXml() {
  const { t1, t2, result } = S;
  const { marks1, marks2, records, summary } = result;

  const border = '<Borders>' +
    '<Border ss:Position="Bottom" ss:LineStyle="Continuous" ss:Weight="1" ss:Color="#D9D9D9"/>' +
    '<Border ss:Position="Left" ss:LineStyle="Continuous" ss:Weight="1" ss:Color="#D9D9D9"/>' +
    '<Border ss:Position="Right" ss:LineStyle="Continuous" ss:Weight="1" ss:Color="#D9D9D9"/>' +
    '<Border ss:Position="Top" ss:LineStyle="Continuous" ss:Weight="1" ss:Color="#D9D9D9"/>' +
    '</Borders>';

  const styles = `<Styles>
    <Style ss:ID="Default" ss:Name="Normal"><Alignment ss:Vertical="Center" ss:WrapText="1"/>

      <Font ss:FontName="微软雅黑" ss:Size="10" ss:Color="#000000"/><Alignment ss:Vertical="Center" ss:WrapText="1"/>${border}</Style>

    `<Row><Cell ss:StyleID="s_norm"><Data ss:Type="String" xml:space="preserve">${escXml(k)}</Data></Cell>

function tableSheet(name, headers, rows, marks, chgStyle, sel) {
  let xml = `<Worksheet ss:Name="${escXml(name)}"><Table>`;
  xml += '<Row>' + headers.map(h =>
    `<Cell ss:StyleID="s_hdr"><Data ss:Type="String" xml:space="preserve">${escXml(h)}</Data></Cell>`
  ).join('') + '</Row>';
  rows.forEach((row, ri) => {
    if (sel && !sel.has(ri)) return;
    let rowXml = '<Row>';
    for (let ci = 0; ci < headers.length; ci++) {
      const m = marks[ri][ci];
      let style = 's_norm';
      if (m && norm(row[ci]) !== '') {
        if (m === 'same') style = 's_same';
        else if (m === 'add' || m === 'chg') style = chgStyle;
        else if (m === 'del') style = 's_yellow';
      }
      rowXml += `<Cell ss:StyleID="${style}"><Data ss:Type="String" xml:space="preserve">${escXml(row[ci])}</Data></Cell>`;
    }
    rowXml += '</Row>';
    xml += rowXml;
  });
  xml += '</Table></Worksheet>';
  return xml;
}
